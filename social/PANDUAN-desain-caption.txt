    # Crop foto ke kotak WxH tanpa memotong kepala. Tidak pernah crop dari tengah vertikal.

            if len(f):
                x, y, w, h = max(f, key=lambda r: r[2] * r[3])
                cx, top, info = x + w / 2, max(0, y - 0.7 * h), 'wajah terdeteksi'
    except Exception:
        pass
    r = im.resize((round(iw * s), round(ih * s)), Image.LANCZOS)
    left = min(max(cx * s - W / 2, 0), r.width - W)          # orang di tengah kotak
    up = min(max(top * s - H * 0.08, 0), r.height - H)       # ruang 8% di atas kepala
    return r.crop((int(left), int(up), int(left) + W, int(up) + H)), info
